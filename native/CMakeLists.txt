cmake_minimum_required(VERSION 3.20)
project(machine_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(machine_kernel SHARED src/kernel.cpp src/program.cpp)
target_include_directories(machine_kernel PUBLIC include)
target_compile_options(machine_kernel PRIVATE -Wall -Wextra -Werror)
add_executable(machine_conformance tests/conformance.cpp)
target_link_libraries(machine_conformance PRIVATE machine_kernel Threads::Threads)
add_executable(machine_program_conformance tests/program.cpp)
target_link_libraries(machine_program_conformance PRIVATE machine_kernel)
add_executable(machine_benchmark src/benchmark.cpp)
target_link_libraries(machine_benchmark PRIVATE machine_kernel Threads::Threads)
enable_testing()
add_test(NAME native_conformance COMMAND machine_conformance)
add_test(NAME program_conformance COMMAND machine_program_conformance)
add_library(machine_economics SHARED src/economics.cpp)
target_include_directories(machine_economics PUBLIC include)
target_compile_options(machine_economics PRIVATE -Wall -Wextra -Werror)
add_executable(machine_economics_conformance tests/economics.cpp)
target_link_libraries(machine_economics_conformance PRIVATE machine_economics)
add_test(NAME economic_conformance COMMAND machine_economics_conformance)
