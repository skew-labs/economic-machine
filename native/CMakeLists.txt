cmake_minimum_required(VERSION 3.20)
project(machine_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(machine_pump_signal SHARED src/pump_signal.cpp)
target_include_directories(machine_pump_signal PUBLIC include)
target_compile_options(machine_pump_signal PRIVATE -Wall -Wextra -Werror)
add_library(machine_kernel SHARED src/kernel.cpp src/program.cpp)
target_include_directories(machine_kernel PUBLIC include)
target_compile_options(machine_kernel PRIVATE -Wall -Wextra -Werror)
add_executable(machine_conformance tests/conformance.cpp)
target_link_libraries(machine_conformance PRIVATE machine_kernel Threads::Threads)
add_executable(machine_program_conformance tests/program.cpp)
target_link_libraries(machine_program_conformance PRIVATE machine_kernel)
add_executable(machine_benchmark src/benchmark.cpp)
target_link_libraries(machine_benchmark PRIVATE machine_kernel Threads::Threads)
enable_testing()
add_executable(machine_pump_conformance tests/pump_signal.cpp)
target_include_directories(machine_pump_conformance PRIVATE include)
target_compile_options(machine_pump_conformance PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME pump_conformance COMMAND machine_pump_conformance)
add_executable(machine_sol_mm_conformance tests/sol_mm.cpp)
target_include_directories(machine_sol_mm_conformance PRIVATE include)
target_compile_options(machine_sol_mm_conformance PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME sol_mm_conformance COMMAND machine_sol_mm_conformance)
add_executable(machine_muse_conformance tests/muse_maker.cpp)
target_include_directories(machine_muse_conformance PRIVATE include)
target_compile_options(machine_muse_conformance PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME muse_conformance COMMAND machine_muse_conformance)
add_library(machine_muse_learning SHARED ../products/pacifica_muse/native/learning.cpp)
target_include_directories(machine_muse_learning PRIVATE include)
add_test(NAME native_conformance COMMAND machine_conformance)
add_test(NAME program_conformance COMMAND machine_program_conformance)
add_library(machine_economics SHARED src/economics.cpp)
target_include_directories(machine_economics PUBLIC include)
target_compile_options(machine_economics PRIVATE -Wall -Wextra -Werror)
add_library(machine_resident_gate SHARED src/resident_gate.cpp)
target_include_directories(machine_resident_gate PUBLIC include)
target_compile_options(machine_resident_gate PRIVATE -Wall -Wextra -Werror)
add_executable(machine_economics_conformance tests/economics.cpp)
target_link_libraries(machine_economics_conformance PRIVATE machine_economics)
add_test(NAME economic_conformance COMMAND machine_economics_conformance)

# All reusable native tools build through the same portable entry point.
add_library(machine_mining SHARED src/mining.cpp)
target_include_directories(machine_mining PUBLIC include)
add_library(machine_solution SHARED src/solution.cpp)
target_include_directories(machine_solution PUBLIC include)
add_executable(machine_mining_conformance tests/mining.cpp)
target_include_directories(machine_mining_conformance PRIVATE include)
add_executable(machine_solution_conformance tests/solution.cpp)
target_include_directories(machine_solution_conformance PRIVATE include)
add_test(NAME mining_conformance COMMAND machine_mining_conformance)
add_test(NAME solution_conformance COMMAND machine_solution_conformance)

# The sandbox depends on Linux seccomp; do not silently publish an unsandboxed
# pipeline on other platforms. The pure libraries above remain portable.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(machine_pump_probe src/pump_probe.cpp)
  target_include_directories(machine_pump_probe PRIVATE include)
  target_compile_options(machine_pump_probe PRIVATE -Wall -Wextra -Werror)
  add_executable(machine_solution_pipeline src/solution_pipeline.cpp)
  target_link_libraries(machine_solution_pipeline PRIVATE Threads::Threads)
  add_executable(machine_pipeline_conformance tests/solution_pipeline.cpp)
  target_link_libraries(machine_pipeline_conformance PRIVATE Threads::Threads)
  add_test(NAME pipeline_conformance COMMAND machine_pipeline_conformance)
  add_test(NAME pipeline_sandbox COMMAND machine_solution_pipeline --sandbox-self-test)
endif()

# The conformance programs use assert(). Keep their checks enabled even when
# users choose Release; otherwise CTest could return a false pass.
foreach(target machine_conformance machine_program_conformance machine_economics_conformance
               machine_mining_conformance machine_solution_conformance)
  target_compile_options(${target} PRIVATE -UNDEBUG)
endforeach()
if(TARGET machine_pipeline_conformance)
  target_compile_options(machine_pipeline_conformance PRIVATE -UNDEBUG)
endif()
